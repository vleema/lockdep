\section{Introdução e Objetivo}
\begin{frame}{Objetivo Geral}
  \begin{itemize}
    \item Estender mutexes didáticos para detectar deadlocks e violações de ordem.
    \item Inspirado no lockdep do Linux.
    \item Abordagens: Grafo de espera (reativo) ou verificação de ordem (preventivo).
  \end{itemize}
\end{frame}

\begin{frame}{Requisitos Principais}
  \begin{itemize}
    \item Base em mutex C.
    \item Estruturas: Dono atual, fila de espera, ordem por thread.
    \item Implementar pelo menos uma abordagem (extra para ambas).
    \item Foco aqui: Lockdep-style (preventivo); grafo de espera como extensão.
  \end{itemize}
\end{frame}

\section{Fundamentos Teóricos}
\begin{frame}{Conceitos Chave}
  \begin{itemize}
    \item Deadlock: Espera circular (condições de Coffman).
    \item Grafo de Espera: Nodes=threads/resources; ciclo=deadlock real (DFS).
    \item Lockdep-Style: Grafo dependências (nodes=locks, edges=ordem); ciclo=potencial inversão.
  \end{itemize}
\end{frame}

\section{Implementação por Arquivo}
\begin{frame}{lockdep.h: Estruturas de Dados}
  \begin{itemize}
    \item lock_node_t: Grafo (children) + lista (next); dono (current_held_lock).
    \item adjacency_locks_t: Edges para dependências.
    \item held_lock_t: Pilha de holds com trace e timestamp.
    \item thread_context_t: Contexto thread com pilha held.
  \end{itemize}

\begin{frame}{pthread_interpose.c: Hooking Pthread}
  \begin{itemize}
    \item Interposição via dlsym para lock/unlock/trylock.
    \item Chama lockdep_acquire/release; evita recursão (in_interpose).
    \item Para trylock: Prevene se deadlock detectado.
  \end{itemize}
  \lstinputlisting[language=C, firstline=30, lastline=40]{\cInterpose} % Trecho chave
\end{frame}

\begin{frame}{lockdep_core.c: Lógica Central}
  \begin{itemize}
    \item acquire_lock: Registra, adiciona dep, checa ciclo (DFS).
    \item has_cycle: DFS recursivo para detectar ciclos; reporta com traces.
    \item print_cycle: Relatório detalhado (path, owners, sugestão order).
  \end{itemize}

\section{Conclusão e Extensões}
\begin{frame}{Conclusão}
  \begin{itemize}
    \item Atende lockdep-style; demonstra grafos em concurrency.
    \item Limitações: O(n) lookups (add hash); sem fila wait full.
    \item Impacto: Educacional; usa IA para análise.
  \end{itemize}
\end{frame}
